# Compile the actual Linux callback body with narrow PipeWire/transport stubs.
# This checks dispatch and copy safety, not Linux ABI or live graph scheduling.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PWAR_SOURCE_DIR}/linux/libpwar.c")
file(READ "${PWAR_SOURCE_DIR}/linux/libpwar.c" PWAR_LINUX_SOURCE)
string(FIND "${PWAR_LINUX_SOURCE}" "static void on_process(void *userdata, struct spa_io_position *position) {" CALLBACK_START)
string(FIND "${PWAR_LINUX_SOURCE}" "static const struct pw_filter_events" CALLBACK_END)
if(CALLBACK_START LESS 0 OR CALLBACK_END LESS CALLBACK_START)
    message(FATAL_ERROR "Cannot locate Linux callback for qualification")
endif()
math(EXPR CALLBACK_LENGTH "${CALLBACK_END} - ${CALLBACK_START}")
string(SUBSTRING "${PWAR_LINUX_SOURCE}" ${CALLBACK_START} ${CALLBACK_LENGTH} CALLBACK_SOURCE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linux_callback.inc" "${CALLBACK_SOURCE}")
if(CALLBACK_SOURCE MATCHES "pwar_audio_queue_write")
    if(UNIX)
        add_executable(PwarLinuxCallbackTests linux_queue_callback_tests.c
            "${PWAR_SOURCE_DIR}/protocol/pwar_audio_queue.c")
        set_property(TARGET PwarLinuxCallbackTests PROPERTY C_STANDARD 11)
    else()
        message(STATUS "Queued Linux callback requires the Linux C11 qualification build")
    endif()
else()
    add_executable(PwarLinuxCallbackTests linux_callback_tests.cpp)
endif()
if(TARGET PwarLinuxCallbackTests)
target_include_directories(PwarLinuxCallbackTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${PWAR_SOURCE_DIR}/protocol")
if(MSVC)
    target_compile_features(PwarLinuxCallbackTests PRIVATE cxx_std_17)
    target_compile_options(PwarLinuxCallbackTests PRIVATE /W4 /EHsc)
endif()
add_test(NAME PwarLinuxCallbackTests COMMAND PwarLinuxCallbackTests)
set_tests_properties(PwarLinuxCallbackTests PROPERTIES TIMEOUT 10)
endif()
