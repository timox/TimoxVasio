cmake_minimum_required(VERSION 3.16)
project(NativeAudioSchannelProbe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Schannel feasibility probe requires Windows")
endif()
enable_testing()
option(TIMOX_SCHANNEL_TEMPORARY_USER_KEY_TEST "Enable explicit temporary user-key TLS test" OFF)
add_executable(NativeAudioSchannelProbe probe.cpp)
target_compile_features(NativeAudioSchannelProbe PRIVATE cxx_std_17)
target_compile_definitions(NativeAudioSchannelProbe PRIVATE SECURITY_WIN32 NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NativeAudioSchannelProbe PRIVATE /W4 /WX /EHsc)
target_link_libraries(NativeAudioSchannelProbe PRIVATE secur32 crypt32 ncrypt bcrypt)
add_test(NAME NativeAudioSchannelProbe COMMAND NativeAudioSchannelProbe)
set_tests_properties(NativeAudioSchannelProbe PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
if(TIMOX_SCHANNEL_TEMPORARY_USER_KEY_TEST)
    add_test(NAME NativeAudioSchannelTemporaryUserKey COMMAND NativeAudioSchannelProbe --temporary-user-key)
    set_tests_properties(NativeAudioSchannelTemporaryUserKey PROPERTIES TIMEOUT 120)
endif()
