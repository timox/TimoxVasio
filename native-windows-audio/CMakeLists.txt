cmake_minimum_required(VERSION 3.16)
project(TimoxNativeAudioProtocol LANGUAGES C CXX)
enable_testing()
add_executable(NativeAudioProtocolTests tests/native_audio_protocol_tests.cpp tests/protocol_c_compile.c)
target_include_directories(NativeAudioProtocolTests PRIVATE include)
target_compile_features(NativeAudioProtocolTests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(NativeAudioProtocolTests PRIVATE /W4 /WX)
else()
    target_compile_options(NativeAudioProtocolTests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME NativeAudioProtocolTests COMMAND NativeAudioProtocolTests)
set_tests_properties(NativeAudioProtocolTests PROPERTIES TIMEOUT 10)
