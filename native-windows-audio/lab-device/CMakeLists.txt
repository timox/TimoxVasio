cmake_minimum_required(VERSION 3.16)
project(TimoxNativeLabDevice LANGUAGES CXX)
enable_testing()
add_executable(NativeLabDeviceTransactionTests tests/device_transaction_tests.cpp device_transaction.cpp)
target_include_directories(NativeLabDeviceTransactionTests PRIVATE .)
target_compile_features(NativeLabDeviceTransactionTests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(NativeLabDeviceTransactionTests PRIVATE /W4 /WX /EHsc)
    set_property(TARGET NativeLabDeviceTransactionTests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
else()
    target_compile_options(NativeLabDeviceTransactionTests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME NativeLabDeviceTransactionTests COMMAND NativeLabDeviceTransactionTests)
set_tests_properties(NativeLabDeviceTransactionTests PROPERTIES TIMEOUT 15)
foreach(component lab_cli receipt_codec receipt_file_policy)
    add_executable(${component}_tests tests/${component}_tests.cpp ${component}.cpp)
    target_include_directories(${component}_tests PRIVATE .)
    target_compile_features(${component}_tests PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(${component}_tests PRIVATE /W4 /WX /EHsc)
        set_property(TARGET ${component}_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    else()
        target_compile_options(${component}_tests PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME ${component}_tests COMMAND ${component}_tests)
endforeach()
# Native mutation is never invoked by these tests.
if(WIN32)
    add_executable(TimoxNativeLabDevice main.cpp device_win32.cpp device_transaction.cpp lab_cli.cpp receipt_codec.cpp receipt_file_policy.cpp)
    target_compile_features(TimoxNativeLabDevice PRIVATE cxx_std_17)
    target_compile_definitions(TimoxNativeLabDevice PRIVATE UNICODE _UNICODE)
    target_compile_options(TimoxNativeLabDevice PRIVATE /W4 /WX /EHsc)
    set_property(TARGET TimoxNativeLabDevice PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    target_link_libraries(TimoxNativeLabDevice PRIVATE setupapi advapi32 shell32 ole32)
    add_test(NAME NativeLabDeviceHelp COMMAND TimoxNativeLabDevice --help)
    set_tests_properties(NativeLabDeviceHelp PROPERTIES TIMEOUT 10)
endif()
