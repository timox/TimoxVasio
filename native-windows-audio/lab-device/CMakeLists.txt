cmake_minimum_required(VERSION 3.16)
project(TimoxNativeLabDevice LANGUAGES CXX)
enable_testing()
add_executable(NativeLabDeviceTransactionTests tests/device_transaction_tests.cpp device_transaction.cpp)
target_include_directories(NativeLabDeviceTransactionTests PRIVATE .)
target_compile_features(NativeLabDeviceTransactionTests PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(NativeLabDeviceTransactionTests PRIVATE /W4 /WX /EHsc)
    set_property(TARGET NativeLabDeviceTransactionTests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
else()
    target_compile_options(NativeLabDeviceTransactionTests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME NativeLabDeviceTransactionTests COMMAND NativeLabDeviceTransactionTests)
set_tests_properties(NativeLabDeviceTransactionTests PROPERTIES TIMEOUT 15)
